  const def = this.registry.get(paneId)
  if (def == null || def.scope === 'operator') return false      // `not-a-pane` / `operator-scope`
  return this.registry.isEnabled(paneId)                         // `not-enabled`

async paintReplumbedPane(paneId: string): Promise<void> {
  if (paneId === 'crosslinks') { /* B-5's arm, byte-for-byte — NOT gated, not touched */ }
  if (paneId === 'doc-nav' && this.replumbedPaneCanRender(paneId)) {
    const generation = (this.docHeadsGeneration += 1)    // CAPTURE before the await
    try {
      const answer = await this.bridge.rag.docHeads()      // ONE call and NOTHING else
      if (generation !== this.docHeadsGeneration) return // superseded: no commit, no paint

      if (generation !== this.docHeadsGeneration) return // the same rule inside the catch

  this.updatePane(paneId)                                  // B-3's ONE paint, unchanged
